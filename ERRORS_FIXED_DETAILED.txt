  │         '%Y-%m-%d %H:%M').date()                                   │

  │         appointment_time_str, '%H:%M').time()                      │
  │                                                                     │
  │     # Create appointment                                            │
  │     appointment = Appointment(                                      │
  │         patient_id=patient.id,                                     │
